# Single Prospector receiver, two ESB keyboard halves. No touch/menu/HID bridge.
if(CONFIG_LISM_SPLIT_ESB OR CONFIG_LISM_PROSPECTOR_LOCK_INDICATORS)
  zephyr_library()
  include("${CMAKE_CURRENT_LIST_DIR}/cmake/lism_charge_irq.cmake")
  zephyr_library_sources_ifdef(CONFIG_LISM_PROSPECTOR_LOCK_INDICATORS src/prospector_lock_indicators.c)
  if(CONFIG_LISM_SPLIT_ESB)
    if(CONFIG_MPSL_DYNAMIC_INTERRUPTS)
      message(FATAL_ERROR "MPSL must own static RADIO/TIMER0/RTC0 vectors")
    endif()
    include("${CMAKE_CURRENT_LIST_DIR}/cmake/lism_esb.cmake")
    lism_esb_prepare_sdk("${CMAKE_CURRENT_BINARY_DIR}/lism_nordic_esb.c")
    zephyr_library_sources(src/esb/protocol.c src/esb/codec.c src/esb/radio.c)
    if(CONFIG_ZMK_SPLIT_ROLE_CENTRAL)
      if(NOT CONFIG_ZMK_SPLIT_BLE_CENTRAL_PERIPHERALS EQUAL 2)
        message(FATAL_ERROR "Only left/right split slots allowed")
      endif()
      include("${CMAKE_CURRENT_LIST_DIR}/cmake/lism_output_behavior.cmake")
      set(outputs "${APPLICATION_SOURCE_DIR}/src/behaviors/behavior_outputs.c")
      lism_patch_output_behavior("${outputs}" "${CMAKE_CURRENT_BINARY_DIR}/lism_behavior_outputs.c")
      set_source_files_properties("${outputs}" TARGET_DIRECTORY app PROPERTIES HEADER_FILE_ONLY ON)
      zephyr_library_sources("${CMAKE_CURRENT_BINARY_DIR}/lism_behavior_outputs.c")
      set(ep "${APPLICATION_SOURCE_DIR}/src/endpoints.c")
      lism_hid_patch_endpoints("${ep}" "${CMAKE_CURRENT_BINARY_DIR}/legacy_endpoints.c")
      set_source_files_properties("${ep}" TARGET_DIRECTORY app PROPERTIES HEADER_FILE_ONLY ON)
      zephyr_library_sources("${CMAKE_CURRENT_BINARY_DIR}/legacy_endpoints.c")
      include("${CMAKE_CURRENT_LIST_DIR}/cmake/lism_split_sets.cmake")
      set(central "${APPLICATION_SOURCE_DIR}/src/split/bluetooth/central.c")
      set(ble "${APPLICATION_SOURCE_DIR}/src/ble.c")
      set(central_copy "${CMAKE_CURRENT_BINARY_DIR}/legacy_central.c")
      set(ble_copy "${CMAKE_CURRENT_BINARY_DIR}/legacy_ble.c")
      lism_hid_patch_central("${central}" "${central_copy}" "${CMAKE_CURRENT_LIST_DIR}/src/lism_hid_host.h")
      lism_hid_patch_ble("${ble}" "${ble_copy}")
      lism_split_sets_patch_ble("${ble_copy}" "${ble_copy}" "${CMAKE_CURRENT_LIST_DIR}/src/lism_split_sets.h")
      lism_split_sets_patch_central("${central_copy}" "${central_copy}" "${CMAKE_CURRENT_LIST_DIR}/src/lism_split_sets.h")
      lism_esb_patch_central("${central_copy}" "${CMAKE_CURRENT_LIST_DIR}/src/esb/integration.h")
      set(input_split "${APPLICATION_SOURCE_DIR}/src/pointing/input_split.c")
      set(input_core "${ZEPHYR_BASE}/subsys/input/input.c")
      lism_split_sets_patch_input_split("${input_split}" "${CMAKE_CURRENT_BINARY_DIR}/legacy_input_split.c")
      lism_split_sets_patch_input_core("${input_core}" "${CMAKE_CURRENT_BINARY_DIR}/legacy_input_core.c")
      set_source_files_properties("${central}" "${ble}" "${input_split}" TARGET_DIRECTORY app PROPERTIES HEADER_FILE_ONLY ON)
      set_source_files_properties("${input_core}" DIRECTORY "${ZEPHYR_BASE}/subsys/input" PROPERTIES HEADER_FILE_ONLY ON)
      zephyr_library_sources("${central_copy}" "${ble_copy}" "${CMAKE_CURRENT_BINARY_DIR}/legacy_input_split.c" "${CMAKE_CURRENT_BINARY_DIR}/legacy_input_core.c" src/lism_split_sets.c src/esb/central.c)
    else()
      include("${CMAKE_CURRENT_LIST_DIR}/cmake/lism_split_role.cmake")
      set(role_src "${APPLICATION_SOURCE_DIR}/src/split/bluetooth/peripheral.c")
      lism_split_patch_role("${role_src}" "${CMAKE_CURRENT_BINARY_DIR}/legacy_role_peripheral.c" "${CMAKE_CURRENT_LIST_DIR}/src/lism_split_role.h")
      set_source_files_properties("${role_src}" TARGET_DIRECTORY app PROPERTIES HEADER_FILE_ONLY ON)
      zephyr_library_include_directories("${APPLICATION_SOURCE_DIR}/src/split/bluetooth")
      zephyr_library_sources("${CMAKE_CURRENT_BINARY_DIR}/legacy_role_peripheral.c" src/esb/peripheral.c)
    endif()
  endif()
endif()
